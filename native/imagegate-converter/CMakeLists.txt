cmake_minimum_required(VERSION 3.24)
project(imagegate_converter VERSION 0.1.0 LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
# Candidate source baseline, not deployment approval. No FetchContent or dependency downloads.
pkg_check_modules(VIPS REQUIRED IMPORTED_TARGET vips=8.18.7)
add_executable(imagegate-converter src/main.cpp)
target_compile_features(imagegate-converter PRIVATE cxx_std_17)
target_link_libraries(imagegate-converter PRIVATE PkgConfig::VIPS)
if(MSVC)
  target_compile_options(imagegate-converter PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(imagegate-converter PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(IMAGEGATE_BUILD_RESOURCE_TESTS "Build test-only Windows resource probe" OFF)
option(IMAGEGATE_BUILD_SANDBOX "Build experimental Windows LPAC parent and contained worker" OFF)
if(IMAGEGATE_BUILD_SANDBOX)
  if(NOT WIN32)
    message(FATAL_ERROR "LPAC sandbox requires Windows")
  endif()
  add_library(imagegate-sandbox STATIC src/sandbox_runner.cpp src/sandbox_lifecycle.cpp)
  target_compile_definitions(imagegate-sandbox PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_compile_features(imagegate-sandbox PUBLIC cxx_std_17)
  target_include_directories(imagegate-sandbox PUBLIC src)
  target_link_libraries(imagegate-sandbox PUBLIC userenv advapi32 ole32 shell32 uuid)
  add_executable(imagegate-sandbox-launcher src/launcher.cpp)
  target_link_libraries(imagegate-sandbox-launcher PRIVATE imagegate-sandbox)
  add_executable(imagegate-contained-worker src/main.cpp)
  target_compile_definitions(imagegate-contained-worker PRIVATE IMAGEGATE_CONTAINED_WORKER=1)
  target_compile_features(imagegate-contained-worker PRIVATE cxx_std_17)
  target_link_libraries(imagegate-contained-worker PRIVATE PkgConfig::VIPS advapi32)
  if(MINGW)
    target_link_options(imagegate-sandbox-launcher PRIVATE -municode -static -Wl,--no-insert-timestamp)
    target_link_options(imagegate-contained-worker PRIVATE -municode -Wl,--no-insert-timestamp)
  endif()
  if(IMAGEGATE_BUILD_RESOURCE_TESTS)
    add_executable(imagegate-access-probe tests/access_probe.cpp)
    target_include_directories(imagegate-access-probe PRIVATE src)
    target_compile_features(imagegate-access-probe PRIVATE cxx_std_17)
    target_compile_definitions(imagegate-access-probe PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(imagegate-access-probe PRIVATE advapi32 ws2_32)
    add_executable(imagegate-network-probe tests/access_probe.cpp)
    target_include_directories(imagegate-network-probe PRIVATE src)
    target_compile_features(imagegate-network-probe PRIVATE cxx_std_17)
    target_compile_definitions(imagegate-network-probe PRIVATE IMAGEGATE_NETWORK_PROBE=1 _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(imagegate-network-probe PRIVATE advapi32 ws2_32)
    add_executable(imagegate-sandbox-test-launcher tests/sandbox_test_launcher.cpp)
    target_link_libraries(imagegate-sandbox-test-launcher PRIVATE imagegate-sandbox)
    if(MINGW)
      target_link_options(imagegate-access-probe PRIVATE -municode -static -Wl,--no-insert-timestamp)
      target_link_options(imagegate-network-probe PRIVATE -municode -static -Wl,--no-insert-timestamp)
      target_link_options(imagegate-sandbox-test-launcher PRIVATE -municode -static -Wl,--no-insert-timestamp)
    endif()
  endif()
endif()
if(MINGW)
  target_link_options(imagegate-converter PRIVATE -municode -Wl,--no-insert-timestamp)
endif()
if(IMAGEGATE_BUILD_RESOURCE_TESTS)
  if(NOT WIN32)
    message(FATAL_ERROR "Resource probes currently require Windows")
  endif()
  add_executable(imagegate-job-probe tests/job_probe.cpp)
  target_compile_features(imagegate-job-probe PRIVATE cxx_std_17)
  target_include_directories(imagegate-job-probe PRIVATE src)
  if(MINGW)
    target_link_options(imagegate-job-probe PRIVATE -municode -Wl,--no-insert-timestamp)
  endif()
endif()
